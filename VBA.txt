Sub Berechnen()
    ThisWorkbook.Save
    Dim Results_Pfad As String
    Dim pic As Picture
    Dim exe_prompt As VbMsgBoxResult
        
    Application.ScreenUpdating = False
    Application.DisplayAlerts = False
    
    Path = ThisWorkbook.Path
    exe_prompt = MsgBox("Wählen Sie Ihre python.exe-Datei.", (vbOKCancel + vbInformation), "WGK-Tool")
    If exe_prompt = vbOK Then
        pythonScriptPath = Application.GetOpenFilename("Executable Files (*.exe), *.exe")
    Else
        MsgBox "Das Fraunhofer WGK-Tool benötigt die Softwarekomponente python.exe von Python by Python Software Foundation, um die Python-Anwendung zu starten und die korrekte Funktion der Python-Datei zu ermöglichen.", vbOKOnly + vbInformation, "WGK-Tool Hilfe"
    End If
    
    py_prompt = MsgBox("Wählen Sie die WGK.py-Datei.", (vbOKCancel + vbInformation), "WGK-Tool")
    If py_prompt = vbOK Then
        pythonFilePath = Application.GetOpenFilename("Python Files (*.py), *.py")
    Else
        MsgBox "Die Python-Datei des Fraunhofer WGK-Tools enthält alle Funktionen und Werte, die notwendig sind, um die Berechnungen der WGK durchzuführen.", vbOKOnly + vbInformation, "WGK-Tool Hilfe"
    End If
    
    If pythonFilePath <> "False" Then
        Shell pythonScriptPath & " " & pythonFilePath, vbNormalFocus
    Else
        MsgBox "Bitte überprüfen Sie die ausgewählten Dateien. Es ist ein Fehler aufgetreten..", vbOKOnly + vbExclamation, "WGK-Tool Hilfe"
    End If
    
End Sub
