Sub Berechnen()
    ThisWorkbook.Save
    Dim Results_Pfad As String
    Dim pic As Picture
        
    Application.ScreenUpdating = False
    Application.DisplayAlerts = False
    
    Path = ThisWorkbook.Path
    
    pythonScriptPath = Application.GetOpenFilename("Executable Files (*.exe), *.exe")
    pythonFilePath = Application.GetOpenFilename("Python Files (*.py), *.py")
    
    If pythonFilePath <> "False" Then
        Shell pythonScriptPath & " " & pythonFilePath, vbNormalFocus
    End If
    
End Sub
